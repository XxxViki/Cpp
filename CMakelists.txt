
#要求cmake最低版本
cmake_minimum_required(VERSION 3.10.0)
project(HelloWorld)

#设置C++11标准
set(CMAKE_CXX_STANDARD 11)

#PROJECT_SOURCE_DIR表示当前CMakeLists.txt所在路径
include_directories(${PROJECT_SOURCE_DIR}/include)

#设置生成的可执行文件存放路径
#EXECUTABLE_OUTPUT_PATH ${PROJECT_SOURCE_DIR}获取根目录
set(EXECUTABLE_OUTPUT_PATH ${PROJECT_SOURCE_DIR}/bin)

#设置源码文件路径
#SRC_LIST变量存放src目录下的所有.cpp文件
aux_source_directory(${PROJECT_SOURCE_DIR}/src SRC_LIST)

#add_executable(
#    hello 
#    ${SRC_LIST}
#)

# 为 hello.cpp 生成 hello.exe
add_executable(hello src/hello.cpp)

# 为 CopyConstrucetor.cpp 生成 CopyConstrucetor.exe
add_executable(CopyConstrucetor src/CopyConstrucetor.cpp)

# 为 inherit.cpp 生成 inherit.exe
add_executable(inherit src/inherit.cpp)

# 为 singleto.cpp 生成 singleto.exe
add_executable(singleto src/singleto.cpp)
